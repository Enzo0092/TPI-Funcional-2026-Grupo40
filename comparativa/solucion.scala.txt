// ========================================================
// COMPONENTE: Semaforo
// TIPO: Object / Singleton
// DESCRIPCIÓN: Agrupa las funciones puras transicion y timer
// relacionadas con el comportamiento del semáforo.
// ========================================================

object Semaforo {

  // ========================================================
  // FUNCIÓN: transicion
  // NATURALEZA: Pura (Siempre devuelve el mismo resultado
  // para los mismos parámetros)
  // ESTRATEGIA: Condicional (Pattern Matching mediante match)
  // IMPACTO: No destructiva
  // ========================================================

  def transicion(colorActual: String, cambiarA: String): List[String] = {
    (colorActual, cambiarA) match {
      case ("en-rojo", "en-verde") =>
        List(colorActual, cambiarA)

      case ("en-amarillo", "en-rojo") =>
        List(colorActual, cambiarA)

      case ("en-verde", "en-amarillo") =>
        List(colorActual, cambiarA)

      case _ =>
        Nil
    }
  }

  // ========================================================
  // FUNCIÓN: timer
  // NATURALEZA: Pura (Depende únicamente del parámetro time)
  // ESTRATEGIA: Condicional y Expresión Aritmética
  // IMPACTO: No destructiva
  // ========================================================

  def timer(time: Int): String = {
    val tColor = time % 216

    tColor match {
      case x if x < 90 => "rojo"
      case x if x < 96 => "verde"
      case _ => "amarillo"
    }
  }
}
